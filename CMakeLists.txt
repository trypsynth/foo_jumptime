cmake_minimum_required(VERSION 3.24)
project(foo_jumptime LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Find ATL: try the active compiler's toolset first, then scan all VS installations.
cmake_path(GET CMAKE_CXX_COMPILER PARENT_PATH _cl_dir)
cmake_path(GET _cl_dir PARENT_PATH _cl_dir)
cmake_path(GET _cl_dir PARENT_PATH _cl_dir)
cmake_path(GET _cl_dir PARENT_PATH _msvc_root)
set(_atl_include "")
if(EXISTS "${_msvc_root}/atlmfc/include/atlbase.h")
	set(_atl_include "${_msvc_root}/atlmfc/include")
else()
	file(GLOB _candidates
		"C:/Program Files/Microsoft Visual Studio/*/*/VC/Tools/MSVC/*/atlmfc/include"
		"C:/Program Files (x86)/Microsoft Visual Studio/*/*/VC/Tools/MSVC/*/atlmfc/include"
	)
	foreach(_dir IN LISTS _candidates)
		if(EXISTS "${_dir}/atlbase.h")
			set(_atl_include "${_dir}")
		endif()
	endforeach()
endif()
if(_atl_include)
	cmake_path(GET _atl_include PARENT_PATH _atlmfc_root)
	if(CMAKE_SIZEOF_VOID_P EQUAL 8)
		set(_atl_lib "${_atlmfc_root}/lib/x64")
	else()
		set(_atl_lib "${_atlmfc_root}/lib/x86")
	endif()
	include_directories(SYSTEM "${_atl_include}")
	link_directories("${_atl_lib}")
	message(STATUS "Found ATL: ${_atl_include}")
else()
	message(FATAL_ERROR "ATL not found. Install 'C++ ATL for latest build tools' in the VS Installer.")
endif()
add_subdirectory(foosdk)
set(FOOSDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/foosdk")

set(SOURCES src/main.cpp src/time_parser.cpp src/jump_dialog.cpp src/mainmenu_commands.cpp src/foo_jumptime.rc)
add_library(foo_jumptime MODULE ${SOURCES})
target_include_directories(foo_jumptime PRIVATE ${FOOSDK_DIR}/sdk ${FOOSDK_DIR}/sdk/foobar2000 ${FOOSDK_DIR}/wtl/Include ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(foo_jumptime PRIVATE foosdk)
if("${CMAKE_SIZEOF_VOID_P}" STREQUAL "8")
	target_link_libraries(foo_jumptime PRIVATE "${FOOSDK_DIR}/sdk/foobar2000/shared/shared-x64.lib")
else()
	target_link_libraries(foo_jumptime PRIVATE "${FOOSDK_DIR}/sdk/foobar2000/shared/shared-Win32.lib")
endif()
target_link_libraries(foo_jumptime PRIVATE winmm)
target_compile_definitions(foo_jumptime PRIVATE _UNICODE UNICODE STRICT)
if(MSVC)
	target_compile_options(foo_jumptime PRIVATE /MP /wd4302 /wd4838 /wd4996)
endif()
set_target_properties(foo_jumptime PROPERTIES OUTPUT_NAME "foo_jumptime" SUFFIX ".dll")
